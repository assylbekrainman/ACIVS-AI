#### ГЕНЕРАЦИЯ .docx — ТОЧНЫЙ АЛГОРИТМ (python-docx)

Установка: `pip install python-docx --break-system-packages --quiet`

**Ключевой принцип:** использовать файл `Письмо в КН.docx` как base document.
В нём уже встроен логотип Фонда (P[0]) и строка адреса (P[1]).
Нужно сохранить только эти два элемента, всё остальное — пересоздать.

```python
from docx import Document
from docx.oxml.ns import qn
from docx.oxml import OxmlElement
import copy, glob

# 1. Найти шаблон в папке ГП или загруженных файлах
templates = (glob.glob('/sessions/*/mnt/**/*Письмо в КН*.docx', recursive=True) +
             glob.glob('/sessions/*/mnt/uploads/*Письмо в КН*.docx'))
TEMPLATE_PATH = templates[0] if templates else None
# Если шаблон не найден — сообщить пользователю, что нужен файл "Письмо в КН.docx"

doc = Document(TEMPLATE_PATH)
body = doc.element.body
paras = doc.paragraphs

# 2. Клонировать форматирование из эталонных абзацев шаблона (индексы фиксированы!)
def get_pPr(para):
    pPr = para._p.find(qn('w:pPr'))
    return copy.deepcopy(pPr) if pPr is not None else OxmlElement('w:pPr')

ref_recip      = get_pPr(paras[3])   # адресат: отступ лев.=5812 twips, жирный, 14pt
ref_body       = get_pPr(paras[8])   # тело письма: красная строка=709, justify, 14pt
ref_sign       = get_pPr(paras[22])  # подпись: красная строка=851, курсив, 14pt
ref_annex_hdr  = get_pPr(paras[14])  # «Қосымша:»: красная строка=709, курсив, 11pt
ref_annex_item = get_pPr(paras[16])  # пункт приложения: курсив, 12pt

# 3. Удалить всё тело начиная с P[2] (сохранить P[0]=лого, P[1]=адрес)
sectPr = body.find(qn('w:sectPr'))
for el in list(body)[2:]:
    if el is not sectPr: body.remove(el)

# 4. Вспомогательные функции
def make_para(pPr_clone, text, bold=False, italic=False, sz=None):
    p = OxmlElement('w:p')
    if pPr_clone is not None: p.append(copy.deepcopy(pPr_clone))
    if text:
        r = OxmlElement('w:r'); rPr = OxmlElement('w:rPr')
        fonts = OxmlElement('w:rFonts')
        for a in ('w:ascii','w:hAnsi','w:cs'): fonts.set(qn(a), 'Times New Roman')
        rPr.append(fonts)
        if bold:   rPr.append(OxmlElement('w:b'))
        if italic: rPr.append(OxmlElement('w:i'))
        if sz:
            for tag in ('w:sz','w:szCs'):
                e = OxmlElement(tag); e.set(qn('w:val'), str(sz*2)); rPr.append(e)
        r.append(rPr)
        t = OxmlElement('w:t'); t.text = text
        t.set('{http://www.w3.org/XML/1998/namespace}space', 'preserve')
        r.append(t); p.append(r)
    return p

def empty_para(): return OxmlElement('w:p')

def separator_para():
    p = OxmlElement('w:p'); pPr = OxmlElement('w:pPr')
    pBdr = OxmlElement('w:pBdr'); bot = OxmlElement('w:bottom')
    bot.set(qn('w:val'),'single'); bot.set(qn('w:sz'),'6')
    bot.set(qn('w:space'),'1'); bot.set(qn('w:color'),'000000')
    pBdr.append(bot); pPr.append(pBdr); p.append(pPr); return p

def ins(elem): body.insert(list(body).index(sectPr), elem)

# pPr для исполнителя — без красной строки
pPr_exec = copy.deepcopy(ref_body)
ind = pPr_exec.find(qn('w:ind'))
if ind is not None: ind.set(qn('w:firstLine'), '0')

# 5. КЗ-блок
ins(empty_para())
for line in ["Қазақстан Республикасы","Ғылым және жоғары","білім министрлігінің","Ғылым комитеті"]:
    ins(make_para(ref_recip, line, bold=True, sz=14))
ins(empty_para()); ins(empty_para())
for txt in [P1_KZ, P2_KZ, P3_KZ, P4_KZ, P5_KZ]:   # абзацы из раздела ниже
    ins(make_para(ref_body, txt, sz=14))
ins(empty_para())
ins(make_para(ref_annex_hdr, "Қосымша:", italic=True, sz=11))
for i, txt in enumerate(annexes_kz, 1):
    ins(make_para(ref_annex_item, f"{i}) {txt}", italic=True, sz=12))
ins(empty_para()); ins(empty_para())
ins(make_para(ref_sign,
    "Басқарма төрағасы                                                    Ашкин А.А.", sz=14))
ins(empty_para())
ins(make_para(pPr_exec, "Орынд.: Снадин А. / Тел.: 8-7172-57-50-08", sz=11))

# 6. Разделитель
ins(empty_para()); ins(separator_para()); ins(empty_para())

# 7. РУ-блок (аналогично КЗ)
for line in ["Комитет науки","Министерства науки и высшего","образования Республики Казахстан"]:
    ins(make_para(ref_recip, line, bold=True, sz=14))
ins(empty_para()); ins(empty_para())
for txt in [P1_RU, P2_RU, P3_RU, P4_RU, P5_RU]:
    ins(make_para(ref_body, txt, sz=14))
ins(empty_para())
ins(make_para(ref_annex_hdr, "Приложение:", italic=True, sz=11))
for i, txt in enumerate(annexes_ru, 1):
    ins(make_para(ref_annex_item, f"{i}) {txt}", italic=True, sz=12))
ins(empty_para()); ins(empty_para())
ins(make_para(ref_sign,
    "Председатель Правления                                          А.А. Ашкин", sz=14))

ins(make_para(pPr_exec, "Исп.: Снадин А. / Тел.: 8-7172-57-50-08", sz=11))
```
